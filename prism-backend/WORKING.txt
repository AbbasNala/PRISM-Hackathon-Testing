View history of convo:-

	# 1. Generate contextual suggestions
$body = @{
    session_id = $conversation.session_id
    recognized_signs = @("I", "WANT")
} | ConvertTo-Json

$suggestions = Invoke-RestMethod `
  -Uri http://127.0.0.1:5000/api/suggest `
  -Method POST `
  -ContentType "application/json" `
  -Body $body

Write-Host "`n=== SUGGESTIONS ===" -ForegroundColor Cyan
$suggestions.suggestions

# 2. Confirm a suggested message
$body = @{
    session_id = $conversation.session_id
    confirmed_message = "I want water"
} | ConvertTo-Json

Invoke-RestMethod `
  -Uri http://127.0.0.1:5000/api/confirm `
  -Method POST `
  -ContentType "application/json" `
  -Body $body

# 3. Display the final conversation history
Write-Host "`n=== FINAL CONVERSATION HISTORY ===" -ForegroundColor Yellow

$history = Invoke-RestMethod `
  -Uri "http://127.0.0.1:5000/api/conversation/history/$($conversation.session_id)"

$history.history | Format-Table speaker, text -AutoSize
